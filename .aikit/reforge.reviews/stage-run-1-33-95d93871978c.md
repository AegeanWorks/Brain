+def _mode(report, name):

+def checks(report):
+    """Return the claims this battery's own output contradicts.
+
+    A budget either enforces its own dimension or it does not. These
+    findings check that each capped mode names the right kind, that its
+    usage exceeded its limit, that a capped run is not a success, and that
+    a generous budget left the rule run byte-identical.
+    """
+    findings = []
+    by_mode = {case.get("mode"): case for case in report.get("cases", [])}
+    for mode, kind in EXPECTED_KIND.items():
+        case = by_mode.get(mode)
+        if case is None:
+            findings.append("missing-mode:%s" % mode)
+            continue
+        if case.get("kind") != kind:
+            findings.append("kind:%s:%s!=%s" % (mode, case.get("kind"), kind))
+        if kind is None:

+``evals/execution_budget.py`` prints five modes: a generous budget leaves

+enforcing (or enforced the wrong dimension) only altered the printed JSON.
+
+Measured: generous 13 steps / success / actions match; step raises

+import unittest

+from evals.execution_budget import MODES, checks, run_battery

+#: The dimension each capped mode must name, and the modes that must not raise.
+EXPECTED_KIND = {
+    "generous": None,
+    "step": "steps",
+    "llm": "llm_calls",
+    "token": "tokens",
+    "time": "seconds",
+}

+def _case(report, mode):
+    for case in report["cases"]:
+        if case["mode"] == mode:
+            return case
+    raise AssertionError("mode missing: %s" % mode)
+

+class TestExecutionBudgetChecks(unittest.TestCase):
+    @classmethod
+    def setUpClass(cls):
+        cls.report = run_battery()
+
+    def test_a_real_run_has_no_findings(self):
+        self.assertEqual(checks(self.report), [])

+    def test_each_mode_names_its_own_dimension(self):

+    def test_a_capped_run_exceeds_its_limit(self):
