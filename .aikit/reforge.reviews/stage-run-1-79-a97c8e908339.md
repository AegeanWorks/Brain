+    if "--scripted" not in sys.argv:

+        sys.exit(2)

+"""Rule-based agent loop over GridWorld state dicts.
+
+No learned model and no network calls. The planner is a fixed
+Manhattan-greedy policy that reads only the state dict.
+
+Loop per step in Agent.run:
+  perceive: copy state into Memory.save
+  plan: Planner.plan(state, Memory.recall())
+  act: GridWorld.step(action)
+  observe: Telemetry.log + SelfModel.record + hub_write
+
+Hub: Agent owns a dict. hub_write(content) merges a dict into it.
+get_hub() returns a copy. Modules may hold a getter installed by the
+agent (bind_hub) so all modules can read the same hub mapping.
+"""
+
+import copy
+
+SIZE = 8
+
+
+class Memory:
+    """Ordered store of observed state dicts."""
+
+    def __init__(self):
+        self._items = []
+        self._get_hub = None
+
+    def bind_hub(self, getter):
+        self._get_hub = getter
+
+    def read_hub(self):
+        if self._get_hub is None:
+            return {}
+        return self._get_hub()
+
+    def save(self, state):
+        self._items.append(copy.deepcopy(state))
+        return len(self._items)
+
+    def recall(self, limit=5):
+        if limit is None:
+            return copy.deepcopy(self._items)
+        return copy.deepcopy(self._items[-limit:])
+
+    def clear(self):
+        self._items = []
+
+    def __len__(self):
+        return len(self._items)
+
+
+def _nearest(pos, cells):
+    best = None
+    best_d = None
+    for c in cells:
+        d = abs(c[0] - pos[0]) + abs(c[1] - pos[1])
+        if best_d is None or d < best_d:
+            best_d = d
+            best = c
+    return best
+
+
+def _step_toward(pos, goal, obstacles):
+    dx = goal[0] - pos[0]
+    dy = goal[1] - pos[1]
+    ordered = []
+    # Deterministic axis order: x first, then y.
+    if dx > 0:
+        ordered.append(("right", [pos[0] + 1, pos[1]]))
+    elif dx < 0:
+        ordered.append(("left", [pos[0] - 1, pos[1]]))
+    if dy > 0:
+        ordered.append(("down", [pos[0], pos[1] + 1]))
+    elif dy < 0:
+        ordered.append(("up", [pos[0], pos[1] - 1]))
+    blocked = {tuple(o) for o in (obstacles or [])}
+    for name, nxt in ordered:
+        if 0 <= nxt[0] < SIZE and 0 <= nxt[1] < SIZE and tuple(nxt) not in blocked:
+            return name
+    for name, delta in (("up", [0, -1]), ("down", [0, 1]), ("left", [-1, 0]), ("right", [1, 0])):
+        nxt = [pos[0] + delta[0], pos[1] + delta[1]]
+        if 0 <= nxt[0] < SIZE and 0 <= nxt[1] < SIZE and tuple(nxt) not in blocked:
+            return name
+    return "wait"
+
+
+class Planner:
+    """Fixed rule-based policy over the state dict."""
+
+    def __init__(self):
+        self._get_hub = None
+
+    def bind_hub(self, getter):
+        self._get_hub = getter
+
+    def read_hub(self):
+        if self._get_hub is None:
+            return {}
+        return self._get_hub()
+
+    def plan(self, state, memories=None):
+        _ = memories
+        x, y = state["x"], state["y"]
+        pos = [x, y]
+        steps = state.get("steps", 0)
+        obstacles = list(state.get("obstacles", []))
+        if "obstacle" in state and state["obstacle"] not in obstacles:
+            obstacles.append(state["obstacle"])
+
+        remaining = state.get("items_positions")
+        if remaining:
+            if pos in [list(c) for c in remaining]:
+                return "collect"
+            goal = _nearest(pos, [list(c) for c in remaining])
+            return _step_toward(pos, goal, obstacles)
+
+        target = state.get("target")
+        if target is not None:
+            target = list(target)
+            if pos == target:
+                if state.get("task") == "T2_delayed_match" and steps < 2:
+                    return "wait"
+                return "collect"
+            return _step_toward(pos, target, obstacles)
+
+        return "wait"
+
+
+class SelfModel:
+    """Record of (state, action, reward, done) transitions."""
+
+    def __init__(self):
+        self._history = []
+        self._get_hub = None
+
+    def bind_hub(self, getter):
+        self._get_hub = getter
+
+    def read_hub(self):
+        if self._get_hub is None:
+            return {}
+        return self._get_hub()
+
+    def record(self, state, action, reward, done):
+        self._history.append(
+            {"state": copy.deepcopy(state), "action": action, "reward": reward, "done": done}
+        )
+        return len(self._history)
+
+    def history(self):
+        return copy.deepcopy(self._history)
+
+    def clear(self):
+        self._history = []
+
+
+class Telemetry:
+    """Append-only log of step events."""
+
+    def __init__(self):
+        self._events = []
+        self._get_hub = None
+
+    def bind_hub(self, getter):
+        self._get_hub = getter
+
+    def read_hub(self):
+        if self._get_hub is None:
+            return {}
+        return self._get_hub()
+
+    def log(self, event):
+        self._events.append(copy.deepcopy(dict(event)))
+        return len(self._events)
+
+    def events(self):
+        return copy.deepcopy(self._events)
+
+    def clear(self):
+        self._events = []
+
+
+class Agent:
+    """Closed loop over a GridWorld instance.
+
+    Args:
+        memory: Memory instance or None (a default is created).
+        planner: Planner instance or None.
+        selfmodel: SelfModel instance or None.
+        telemetry: Telemetry instance or None.
+        world: GridWorld instance or None. run_task builds one when
+            a task_id/seed pair is given.
+    """
+
+    def __init__(self, memory=None, planner=None, selfmodel=None, telemetry=None, world=None):
+        self.memory = memory if memory is not None else Memory()
+        self.planner = planner if planner is not None else Planner()
+        self.selfmodel = selfmodel if selfmodel is not None else SelfModel()
+        self.telemetry = telemetry if telemetry is not None else Telemetry()
+        self.world = world
+        self._hub = {}
+        for mod in (self.memory, self.planner, self.selfmodel, self.telemetry):
+            bind = getattr(mod, "bind_hub", None)
+            if callable(bind):
+                bind(self.get_hub)
+
+    # -- hub ----------------------------------------------------------
+
+    def hub_write(self, content):
+        """Merge a dict into the shared hub and return a hub copy."""
+        if not isinstance(content, dict):
+            raise TypeError("hub content must be a dict")
+        self._hub.update(copy.deepcopy(content))
+        return self.get_hub()
+
+    def get_hub(self):
+        """Return a copy of the shared hub mapping."""
+        return copy.deepcopy(self._hub)
+
+    def _reset_episode(self):
+        for mod in (self.memory, self.selfmodel, self.telemetry):
+            clear = getattr(mod, "clear", None)
+            if callable(clear):
+                clear()
+        self._hub = {}
+
+    # -- loop ----------------------------------------------------------
+
+    def run(self, max_steps=200):
+        """Run perceive/plan/act/observe loop; return trace list."""
+        if self.world is None:
+            raise ValueError("Agent.world is None; pass a world or use run_task")
+        if max_steps <= 0:
+            raise ValueError("max_steps must be positive")
+        trace = []
+        state = self.world.reset()
+        for i in range(int(max_steps)):
+            self.memory.save(state)  # perceive
+            memories = self.memory.recall()  # recall for planner
+            action = self.planner.plan(state, memories)  # plan
+            new_state, reward, done = self.world.step(action)  # act
+            self.telemetry.log({"step": i, "action": action, "reward": reward, "done": done})
+            self.selfmodel.record(new_state, action, reward, done)
+            self.hub_write({"last_step": i, "last_action": action, "last_reward": reward})
+            trace.append(
+                {"step": i, "state": copy.deepcopy(new_state), "action": action,
+                 "reward": reward, "done": done}
+            )
+            state = new_state
+            if done:
+                break
+        return trace
+
+    def run_task(self, task_id, seed, max_steps=200):
+        """Build GridWorld(task_id, seed) and run the loop."""
+        from worlds.grid import GridWorld
+
+        self.world = GridWorld(task_id, seed)
+        self._reset_episode()
+        return self.run(max_steps=max_steps)

+"""Stdlib-only estimators for Brain evals.
+
+Conventions:
+- trace: sequence of rows, each row a sequence of floats, shape [T, N].
+- target: sequence of scalars aligned with trace rows.
+- All functions are deterministic given inputs; the only randomness is
+  the seeded generator inside permutation_null.
+"""
+import math
+import random
+
+
+def _discretize(values, bins):
+    xs = [float(v) for v in values]
+    if not xs or bins < 1:
+        return [0] * len(xs)
+    lo, hi = min(xs), max(xs)
+    if hi == lo:
+        return [0] * len(xs)
+    out = []
+    for v in xs:
+        b = int((v - lo) / (hi - lo) * bins)
+        out.append(min(bins - 1, max(0, b)))
+    return out
+
+
+def mutual_info_discrete(xs, ys, bins=8):
+    """Mutual information in nats between discretized xs and ys."""
+    if len(xs) != len(ys) or not xs:
+        return 0.0
+    dx, dy = _discretize(xs, bins), _discretize(ys, bins)
+    n = len(dx)
+    cx = [0] * bins
+    cy = [0] * bins
+    jn = [[0] * bins for _ in range(bins)]
+    for a, b in zip(dx, dy):
+        cx[a] += 1
+        cy[b] += 1
+        jn[a][b] += 1
+    mi = 0.0
+    for a in range(bins):
+        for b in range(bins):
+            pxy = jn[a][b] / n
+            if pxy > 0.0 and cx[a] > 0 and cy[b] > 0:
+                mi += pxy * math.log(pxy / ((cx[a] / n) * (cy[b] / n)))
+    return max(0.0, mi)
+
+
+def _solve(A, b):
+    n = len(A)
+    M = [list(A[i]) + [b[i]] for i in range(n)]
+    for c in range(n):
+        p = max(range(c, n), key=lambda r: abs(M[r][c]))
+        if abs(M[p][c]) < 1e-12:
+            return [0.0] * n
+        M[c], M[p] = M[p], M[c]
+        d = M[c][c]
+        M[c] = [v / d for v in M[c]]
+        for r in range(n):
+            if r != c and M[r][c] != 0.0:
+                f = M[r][c]
+                M[r] = [rv - f * cv for rv, cv in zip(M[r], M[c])]
+    return [row[n] for row in M]
+
+
+def phi_ar(trace):
+    """Residual-variance ratio: 1 - MSE(whole)/MSE(partitioned), clipped at 0."""
+    T = len(trace)
+    if T < 3:
+        return 0.0
+    X = [[float(v) for v in row] for row in trace]
+    n = len(X[0])
+    if n < 1 or any(len(r) != n for r in X):
+        return 0.0
+    XA = [[1.0] + X[t] for t in range(T - 1)]
+    Y = X[1:]
+    p = n + 1
+    XtX = [[0.0] * p for _ in range(p)]
+    for r in XA:
+        for i in range(p):
+            for j in range(p):
+                XtX[i][j] += r[i] * r[j]
+    mse_w, mse_p = 0.0, 0.0
+    for j in range(n):
+        col = [Y[t][j] for t in range(T - 1)]
+        Xty = [sum(XA[t][i] * col[t] for t in range(T - 1)) for i in range(p)]
+        w = _solve(XtX, Xty)
+        resid = 0.0
+        for t in range(T - 1):
+            pred = sum(w[i] * XA[t][i] for i in range(p))
+            resid += (col[t] - pred) ** 2
+        mse_w += resid / (T - 1)
+        xj = [X[t][j] for t in range(T - 1)]
+        mx, my = sum(xj) / len(xj), sum(col) / len(col)
+        sxx = sum((v - mx) ** 2 for v in xj)
+        if sxx < 1e-12:
+            rp = sum((v - my) ** 2 for v in col) / len(col)
+        else:
+            sxy = sum((xj[k] - mx) * (col[k] - my) for k in range(len(col)))
+            a = sxy / sxx
+            rp = sum((col[k] - (my + a * (xj[k] - mx))) ** 2 for k in range(len(col))) / len(col)
+        mse_p += rp
+    if mse_p <= 1e-12:
+        return 0.0
+    return max(0.0, 1.0 - (mse_w / mse_p))
+
+
+def synergy(trace, target):
+    """Joint-minus-marginals gap: (syn, syn_n)."""
+    if not trace or len(trace) != len(target):
+        return (0.0, 0.0)
+    c0 = [float(r[0]) for r in trace]
+    c1 = [float(r[1]) for r in trace] if len(trace[0]) > 1 else [0.0] * len(trace)
+    tj = [float(c0[i]) * 8.0 + float(_discretize(c1, 8)[i]) for i in range(len(c0))]
+    i1 = mutual_info_discrete(c0, list(target))
+    i2 = mutual_info_discrete(c1, list(target))
+    ij = mutual_info_discrete(tj, list(target), bins=64)
+    syn = ij - i1 - i2
+    return (syn, (syn / ij) if ij > 1e-12 else 0.0)
+
+
+def gbi(pre_decodable, post_decodable):
+    """Headroom-normalized change: (post - pre) / (1 - pre)."""
+    pre, post = float(pre_decodable), float(post_decodable)
+    if pre >= 1.0:
+        return 0.0 if post >= pre else post - pre
+    return (post - pre) / (1.0 - pre)
+
+
+def logistic_k(x, y):
+    """Grid search over slope k; returns (best_k, r2)."""
+    xs = [float(v) for v in x]
+    ys = [float(v) for v in y]
+    if len(xs) != len(ys) or not xs:
+        return (0.0, 0.0)
+    s = sorted(xs)
+    x0 = s[len(s) // 2]
+    mean_y = sum(ys) / len(ys)
+    sst = sum((v - mean_y) ** 2 for v in ys)
+    best_k, best_r2 = 0.0, float("-inf")
+    for gi in range(-40, 41):
+        k = gi / 4.0
+        ssr = 0.0
+        for xi, yi in zip(xs, ys):
+            e = math.exp(-k * (xi - x0)) if -k * (xi - x0) < 700 else float("inf")
+            p = 1.0 / (1.0 + e)
+            ssr += (yi - p) ** 2
+        r2 = 1.0 - ssr / sst if sst > 1e-12 else (1.0 if ssr < 1e-12 else 0.0)
+        if r2 > best_r2:
+            best_k, best_r2 = k, r2
+    return (best_k, best_r2)
+
+
+def transfer_gain(naive_steps, reuse_steps):
+    """Relative step reduction: (naive - reuse) / naive."""
+    n, r = float(naive_steps), float(reuse_steps)
+    if n <= 0:
+        return 0.0
+    return (n - r) / n
+
+
+def recovery(time_outside, window):
+    """Window fraction inside bounds: 1 - time_outside / window, clipped."""
+    t, w = float(time_outside), float(window)
+    if w <= 0:
+        return 0.0
+    return min(1.0, max(0.0, 1.0 - t / w))
+
+
+def permutation_null(values_fn, shuffles=200, seed=0):
+    """95th percentile of null statistics from a seeded generator."""
+    rng = random.Random(seed)
+    vals = []
+    for _ in range(max(1, int(shuffles))):
+        try:
+            v = values_fn(rng)
+        except TypeError:
+            v = values_fn()
+        vals.append(float(v))
+    vals.sort()
+    idx = min(len(vals) - 1, int(math.ceil(0.95 * len(vals))) - 1)
+    return vals[max(0, idx)]

+"""Recompute eval statistics from a JSONL log without trusting self-reports.
+
+Understands telemetry.log schema-2 records (header/action/result/verdict
+with M/P/S/hub vectors). Eval signals travel as action records whose
+``args`` carry raw arrays:
+
+- kind "eval_samples": args {"trace": [[float, ...], ...],
+  "target": [float, ...]}
+- kind "eval_decodability": args {"pre": float, "post": float}
+- kind "eval_transfer": args {"naive_steps": float, "reuse_steps": float}
+- kind "eval_recovery": args {"time_outside": float, "window": float}
+
+Bare records {"trace": [...], "target": [...]} or {"pre","post"} pairs
+are also accepted. Scalar keys such as reported_*, syn, phi, gbi, gain,
+score are never read: every returned value is recomputed from raw
+arrays and vectors via evals.estimators.
+"""
+import argparse
+import json
+
+from evals.estimators import gbi, phi_ar, recovery, synergy, transfer_gain
+
+_EVAL_KINDS = ("eval_samples", "eval_decodability", "eval_transfer", "eval_recovery")
+
+
+def _mean(xs):
+    xs = list(xs)
+    return sum(xs) / len(xs) if xs else 0.0
+
+
+def _numbers(value):
+    if isinstance(value, bool):
+        return None
+    if isinstance(value, (int, float)):
+        return float(value)
+    return None
+
+
+def _vec_row(rec):
+    row = []
+    for key in ("M", "P", "S", "hub"):
+        val = rec.get(key)
+        if isinstance(val, (list, tuple)):
+            row.extend(v for v in (_numbers(x) for x in val) if v is not None)
+        elif isinstance(val, (int, float)) and not isinstance(val, bool):
+            row.append(float(val))
+    return row
+
+
+def _handle_eval_payload(args, acc):
+    trace = args.get("trace")
+    target = args.get("target")
+    if isinstance(trace, list) and trace and isinstance(trace[0], list):
+        rows = [[float(v) for v in r] for r in trace]
+        acc["phi_rows"].extend(rows)
+        if isinstance(target, list) and len(target) == len(trace):
+            acc["syn_rows"].extend(rows)
+            acc["syn_targets"].extend(float(v) for v in target)
+    elif isinstance(trace, list) and isinstance(target, list):
+        if len(trace) == len(target) and trace and not isinstance(trace[0], list):
+            two = [[float(v), 0.0] for v in trace]
+            acc["syn_rows"].extend(two)
+            acc["syn_targets"].extend(float(v) for v in target)
+    if "pre" in args and "post" in args:
+        acc["pres"].append(float(args["pre"]))
+        acc["posts"].append(float(args["post"]))
+    if "naive_steps" in args and "reuse_steps" in args:
+        acc["naive"].append(float(args["naive_steps"]))
+        acc["reuse"].append(float(args["reuse_steps"]))
+    if "time_outside" in args and "window" in args:
+        acc["recs"].append(recovery(args["time_outside"], args["window"]))
+
+
+def recompute(log_path):
+    """Read JSONL log and return dict with syn, phi, gbi, gain, recovery."""
+    acc = {"phi_rows": [], "syn_rows": [], "syn_targets": [], "pres": [],
+           "posts": [], "naive": [], "reuse": [], "recs": []}
+    vec_rows = []
+    n_records = 0
+    with open(log_path, "r", encoding="utf-8") as fh:
+        for line in fh:
+            line = line.strip()
+            if not line:
+                continue
+            try:
+                rec = json.loads(line)
+            except json.JSONDecodeError:
+                continue
+            if not isinstance(rec, dict):
+                continue
+            n_records += 1
+            if rec.get("type") == "action" and rec.get("kind") in _EVAL_KINDS:
+                args = rec.get("args")
+                if isinstance(args, dict):
+                    _handle_eval_payload(args, acc)
+            elif rec.get("type") in ("header", "verdict"):
+                pass
+            elif "trace" in rec or ("pre" in rec and "post" in rec):
+                _handle_eval_payload(rec, acc)
+            elif "naive_steps" in rec or "time_outside" in rec:
+                _handle_eval_payload(rec, acc)
+            row = _vec_row(rec)
+            if row:
+                vec_rows.append(row)
+    phi_rows = acc["phi_rows"] or vec_rows
+    width = max((len(r) for r in phi_rows), default=0)
+    if phi_rows and any(len(r) != width for r in phi_rows):
+        phi_rows = [r + [0.0] * (width - len(r)) for r in phi_rows]
+    syn, syn_n = synergy(acc["syn_rows"], acc["syn_targets"]) if acc["syn_rows"] else (0.0, 0.0)
+    return {
+        "syn": syn,
+        "syn_n": syn_n,
+        "phi": phi_ar(phi_rows) if len(phi_rows) >= 3 else 0.0,
+        "gbi": gbi(_mean(acc["pres"]), _mean(acc["posts"])) if acc["pres"] else 0.0,
+        "gain": transfer_gain(_mean(acc["naive"]), _mean(acc["reuse"])) if acc["naive"] else 0.0,
+        "recovery": _mean(acc["recs"]) if acc["recs"] else 0.0,
+        "n_records": n_records,
+    }
+
+
+def main():
+    ap = argparse.ArgumentParser()
+    ap.add_argument("--log", required=True)
+    args = ap.parse_args()
+    print(json.dumps(recompute(args.log), sort_keys=True))
+
+
+if __name__ == "__main__":
+    main()

+        return len(self._steps) - 1
+

+        if 0 <= self._idx < len(self._steps):
+            return dict(self._steps[self._idx])

+        if 0 <= self._idx < len(self._steps):
+            self._steps[self._idx]["status"] = "ok"
+            self._idx += 1

+"""Deterministic trace transforms for ablation.
+
+Trace model: a trace is a list of step mappings (or a single step
+mapping). Each step mapping may carry keys ``M``, ``P``, ``S`` and
+``hub``. Vector values are sequences of numbers; scalar values are
+single numbers; absent or ``None`` values pass through unchanged.
+
+All functions are pure: inputs are not mutated, outputs are new
+containers, results depend only on arguments.
+"""
+
+import random
+
+_MODULE_KEYS = ("M", "P", "S")
+
+
+def _zero_value(value):
+    if isinstance(value, bool):
+        return 0
+    if isinstance(value, int):
+        return 0
+    if isinstance(value, float):
+        return 0.0
+    return 0
+
+
+def _zero_vector(vec):
+    if isinstance(vec, tuple):
+        return tuple(_zero_value(v) for v in vec)
+    return [_zero_value(v) for v in vec]
+
+
+def _copy_step(step):
+    out = dict(step)
+    for key in ("M", "P", "S", "hub"):
+        if key in out and isinstance(out[key], (list, tuple)):
+            out[key] = list(out[key]) if isinstance(out[key], list) else tuple(out[key])
+    return out
+
+
+def _map_trace(trace, func):
+    if isinstance(trace, dict):
+        return func(_copy_step(trace))
+    if isinstance(trace, (list, tuple)):
+        result = [func(_copy_step(step)) for step in trace]
+        return tuple(result) if isinstance(trace, tuple) else result
+    raise TypeError("trace must be a mapping or a list/tuple of mappings")
+

+    """Return trace copy with every ``hub`` vector replaced by zeros.

+    Vector length and container kind are preserved; ``M``/``P``/``S``
+    and all other keys are copied unchanged. ``None`` or missing
+    ``hub`` values pass through unchanged.
+    """
+
+    def _one(step):
+        hub = step.get("hub")
+        if hub is None or "hub" not in step:
+            return step
+        if isinstance(hub, (int, float)) and not isinstance(hub, bool):
+            step["hub"] = _zero_value(hub)
+            return step
+        if isinstance(hub, bool):
+            step["hub"] = 0
+            return step
+        if isinstance(hub, (list, tuple)):
+            step["hub"] = _zero_vector(hub)
+            return step
+        return step

+            step[mod] = _zero_vector(value)
+            return step

+            step[mod] = _zero_value(value)

+"""Append-only JSONL record store, schema 2.
+
+Record model: every line is one JSON object with keys
+``seq``, ``type``, ``ts``, ``M``, ``P``, ``S``, ``hub`` plus ``schema`` (=2)
+and type-specific payload keys.
+
+Durability: each append writes one line, calls ``flush`` + ``os.fsync``
+on the file descriptor, then fsyncs the containing directory when the
+platform permits. Callers record an action with ``append_action`` before
+executing the corresponding operation, so a crash leaves the intended
+operation in the log.
+"""
+
+import hashlib
+import json
+import os
+import time
+from pathlib import Path
+
+SCHEMA_VERSION = 2
+
+
+def sha256_file(path):
+    """Return hex sha256 of file bytes at ``path``."""
+    h = hashlib.sha256()
+    with open(path, "rb") as f:
+        for chunk in iter(lambda: f.read(65536), b""):
+            h.update(chunk)
+    return h.hexdigest()
+
+
+def _dir_fsync(dir_path):
+    """Fsync directory entry for ``dir_path``; no-op where unsupported."""
+    try:
+        fd = os.open(str(dir_path), os.O_RDONLY)
+    except OSError:
+        return
+    try:
+        try:
+            os.fsync(fd)
+        except OSError:
+            pass
+    finally:
+        os.close(fd)
+
+
+class Telemetry:
+    """JSONL append-only store at ``path``."""
+
+    def __init__(self, path):
+        self.path = Path(path)
+        parent = self.path.parent
+        if str(parent) and str(parent) not in ("", "."):
+            parent.mkdir(parents=True, exist_ok=True)
+
+    def _write_record(self, record):
+        record = dict(record)
+        record.setdefault("schema", SCHEMA_VERSION)
+        for key in ("seq", "type", "ts", "M", "P", "S", "hub"):
+            record.setdefault(key, None)
+        line = json.dumps(record, sort_keys=True) + "\n"
+        self.path.parent.mkdir(parents=True, exist_ok=True)
+        with open(self.path, "a", encoding="utf-8") as f:
+            f.write(line)
+            f.flush()
+            os.fsync(f.fileno())
+        try:
+            _dir_fsync(self.path.parent)
+        except OSError:
+            pass
+        return record

+        sha256 of that file is computed and stored.
+        """
+        if manifest_sha256 is None and manifest_path is not None:
+            manifest_sha256 = sha256_file(manifest_path)
+        return self._write_record(

+                "seq": 0,
+                "type": "header",
+                "ts": time.time(),

+                "budget": budget,
+                "manifest_sha256": manifest_sha256,
+                "M": None,
+                "P": None,
+                "S": None,
+                "hub": None,
+            }
+        )

+        the record is durable prior to execution.
+        """
+        return self._write_record(
+            {
+                "schema": SCHEMA_VERSION,
+                "seq": seq,
+                "type": "action",
+                "ts": time.time(),
+                "kind": kind,
+                "args": args,
+                "M": M,
+                "P": P,
+                "S": S,
+                "hub": hub,
+            }

+        """Record the result for a prior action ``seq``."""
+        return self._write_record(
+            {
+                "schema": SCHEMA_VERSION,
+                "seq": seq,
+                "type": "result",
+                "ts": time.time(),

+            )
+        return result
