+"""One exit-code contract for the eval batteries.
+
+Most batteries already refuse malformed input by raising, but a battery
+that merely *prints* a bad number exits 0, so a scripted gate cannot tell a
+healthy run from a regressed one. Ten shipped batteries had no ``assert``,
+``raise`` or non-zero exit at all; measured, ``python -m evals.telemetry_lock``
+exits 0 both intact and with ``--ablate``, the deliberately broken case.
+
+This module is the shared contract: a battery passes a list of findings and
+exits with ``exit_code_for(findings)``. An empty list is 0; any finding is 1.
+``audit_batteries`` reports which modules have adopted the contract, so the
+gap is visible instead of being rediscovered by reading each file.
+
+Pure functions plus one read-only directory scan; no battery is imported.
+"""
+
+import ast
+from pathlib import Path
+
+#: Exit code for a battery with no findings.
+EXIT_OK = 0
+#: Exit code for a battery reporting at least one failed claim.
+EXIT_FINDING = 1
+
+#: Names a module must reference to count as adopting the contract.
+CONTRACT_MARKERS = ("battery_result", "exit_code_for")
+
+
+def exit_code_for(findings):
+    """Return the process exit code for one battery's findings."""
+    return EXIT_FINDING if findings else EXIT_OK
+
+
+def battery_result(name, findings, payload=None):
+    """Build one battery's report: name, ok flag, findings and payload."""
+    findings = list(findings or [])
+    return {
+        "battery": name,
+        "ok": not findings,
+        "findings": findings,
+        "payload": payload if payload is not None else {},
+    }
+
+
+def format_findings(name, findings):
+    """Render findings as one attributable line each, for stderr."""
+    return "".join("%s: FAIL %s\n" % (name, finding) for finding in findings)
+
+
+def _adopts_contract(source):
+    """True when a module's source references the contract by name."""
+    return any(marker in source for marker in CONTRACT_MARKERS)
+
+
+def _refuses_by_raising(tree):
+    """True when the module contains an explicit ``raise`` statement."""
+    return any(isinstance(node, ast.Raise) for node in ast.walk(tree))
+
+
+def audit_batteries(root):
+    """Report which eval modules adopt the exit-code contract.
+
+    A module counts as covered when its source references the contract or
+    when it can at least refuse bad input by raising. The rest only print,
+    so a regression there is invisible to a scripted gate.
+    """
+    directory = Path(root) / "evals"
+    with_contract, without_contract = [], []
+    if not directory.is_dir():
+        return {"total": 0, "with_contract": [], "without_contract": [],
+                "contract": list(CONTRACT_MARKERS)}
+    for path in sorted(directory.glob("*.py")):
+        if path.stem == "__init__":
+            continue
+        try:
+            source = path.read_text(encoding="utf-8")
+        except OSError:
+            without_contract.append(path.stem)
+            continue
+        covered = _adopts_contract(source)
+        if not covered:
+            try:
+                covered = _refuses_by_raising(ast.parse(source))
+            except SyntaxError:
+                covered = False
+        (with_contract if covered else without_contract).append(path.stem)
+    return {
+        "total": len(with_contract) + len(without_contract),
+        "with_contract": with_contract,
+        "without_contract": without_contract,
+        "contract": list(CONTRACT_MARKERS),
+    }
+
+
+def main():
+    """Print the audit; exit 1 when any battery lacks a self-check."""
+    import sys
+
+    root = Path(__file__).resolve().parent.parent
+    report = audit_batteries(root)
+    print("batteries: %d, with a self-check: %d, none: %d"
+          % (report["total"], len(report["with_contract"]),
+             len(report["without_contract"])))
+    for name in report["without_contract"]:
+        print("  NO SELF-CHECK: %s" % name)
+    sys.exit(exit_code_for(report["without_contract"]))
+
+
+if __name__ == "__main__":
+    main()

+def checks(report):
+    """Compare a battery report with the pre-registered expectations.
+
+    Every documented number is a claim. This returns one finding per
+    disagreement, plus structural findings when a case is missing, a reward
+    moved, or the guard stopped separating itself from its absence.
+    """
+    findings = []
+    by_key = {}
+    for case in report.get("cases", []):
+        key = (case.get("mode"), case.get("task"), case.get("seed"))
+        by_key[key] = case
+    for (mode, task, seed), (steps, calls) in EXPECTED.items():
+        case = by_key.get((mode, task, seed))
+        if case is None:
+            findings.append("missing-case:%s:%s:%d" % (mode, task, seed))
+            continue
+        if case.get("steps") != steps:
+            findings.append("steps:%s:%s:%d:%s!=%s"
+                            % (mode, task, seed, case.get("steps"), steps))
+        if case.get("backend_calls") != calls:
+            findings.append("calls:%s:%s:%d:%s!=%s"
+                            % (mode, task, seed, case.get("backend_calls"),

+"""The header-reader battery must check its own memory claim.
+
+``evals/header_reader.py`` exists to prove one property: reading only
+headers costs a small fixed memory instead of retaining every action. The
+README states it as 100k records at 0.1 MB peak versus 124.2 MB, and the
+battery printed those numbers and exited 0 either way, so a change that
+made ``read_headers`` accumulate would only alter the printed JSON.
+
+``checks(report)`` compares the two modes on the property, not on exact
+bytes: the header reader must return exactly the header count, retain far
+less than the full reader, and both must see the same headers. A ratio
+floor keeps the test stable against allocator noise.
+"""
+
+import unittest
+
+from evals.header_reader import RECORDS, checks, run_battery
+
+#: The header reader must use at least this many times less peak memory.
+MIN_RATIO = 10.0
+
+
+def _mode(report, name):
+    for case in report["cases"]:
+        if case["mode"] == name:
+            return case
+    raise AssertionError("mode missing: %s" % name)
+
+
+class TestHeaderReaderChecks(unittest.TestCase):
+    @classmethod
+    def setUpClass(cls):
+        cls.report = run_battery()
+
+    def test_a_real_run_has_no_findings(self):
+        self.assertEqual(checks(self.report), [])
+
+    def test_header_mode_returns_only_headers(self):
+        headers = _mode(self.report, "headers")
+        self.assertEqual(headers["returned"], 1)
+        self.assertEqual(headers["headers"], 1)
+        self.assertEqual(headers["lines"], RECORDS + 1)
+
+    def test_all_mode_retains_every_record(self):
+        everything = _mode(self.report, "all")
+        self.assertEqual(everything["returned"], RECORDS + 1)
+        self.assertEqual(everything["headers"], 1)
+
+    def test_header_mode_retains_far_less(self):
+        everything = _mode(self.report, "all")
+        headers = _mode(self.report, "headers")
+        ratio = everything["peak_bytes"] / max(headers["peak_bytes"], 1)
+        self.assertGreater(ratio, MIN_RATIO, "ratio %.1fx" % ratio)
+
+    def test_both_modes_see_the_same_headers(self):
+        self.assertEqual(_mode(self.report, "all")["headers"],
+                         _mode(self.report, "headers")["headers"])
+
+    def test_a_header_reader_that_retains_everything_is_named(self):
+        broken = {"cases": [dict(c) for c in self.report["cases"]]}
+        _mode(broken, "headers")["peak_bytes"] = \
+            _mode(broken, "all")["peak_bytes"]
+        findings = checks(broken)
+        self.assertTrue(any("memory" in f or "ratio" in f for f in findings),
+                        findings)
+
+    def test_a_header_reader_that_returns_everything_is_named(self):
+        broken = {"cases": [dict(c) for c in self.report["cases"]]}
+        _mode(broken, "headers")["returned"] = RECORDS + 1
+        findings = checks(broken)
+        self.assertTrue(any("returned" in f for f in findings), findings)
+
+    def test_a_lost_header_is_named(self):
+        broken = {"cases": [dict(c) for c in self.report["cases"]]}
+        _mode(broken, "headers")["headers"] = 0
+        findings = checks(broken)
+        self.assertTrue(any("headers" in f for f in findings), findings)
+
+    def test_a_missing_mode_is_named(self):
+        broken = {"cases": [c for c in self.report["cases"]
+                            if c["mode"] != "headers"]}
+        findings = checks(broken)
+        self.assertTrue(any("missing" in f for f in findings), findings)
+
+
+if __name__ == "__main__":
+    unittest.main()
